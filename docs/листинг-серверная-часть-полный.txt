     1	<?php
     2	/**
     3	 * index.php - Единая точка входа (Front Controller)
     4	 *
     5	 * ВСЕ запросы к сайту проходят через этот файл.
     6	 * .htaccess перенаправляет сюда любой URL, который не является
     7	 * реальным файлом или папкой.
     8	 *
     9	 * Задачи этого файла:
    10	 *  1. Настроить окружение (сессия, заголовки безопасности)
    11	 *  2. Разобрать URL и определить нужный контроллер + метод
    12	 *  3. Вызвать контроллер и вернуть ответ
    13	 *
    14	 * Маршруты:
    15	 *   GET  /                          → Главная страница
    16	 *   GET  /catalog                   → Хаб категорий
    17	 *   GET  /catalog/{categoryId}      → Товары категории
    18	 *   GET  /product/{productId}       → Карточка товара с конфигуратором
    19	 *   GET  /api/product/{id}          → JSON: данные товара
    20	 *   GET  /api/csrf-token            → JSON: CSRF-токен
    21	 *   POST /api/order/create          → JSON: создание заказа
    22	 *   POST /api/lead/create           → JSON: заявка на КП
    23	 *   GET  /order/success/{orderId}   → Страница подтверждения заказа
    24	 *   GET  /cart                      → Корзина
    25	 *   GET  /privacy                   → Политика конфиденциальности
    26	 */
    27	
    28	declare(strict_types=1);
    29	
    30	// ------------------------------------------------------------------
    31	// Автозагрузка классов (PSR-4 упрощённый вариант без Composer)
    32	// ------------------------------------------------------------------
    33	spl_autoload_register(function (string $className): void {
    34	    // Преобразуем namespace в путь к файлу:
    35	    // App\Controllers\OrderController → /controllers/OrderController.php
    36	    $prefix   = 'App\\';
    37	    $baseDir  = __DIR__ . '/';
    38	
    39	    if (str_starts_with($className, $prefix)) {
    40	        $relative = substr($className, strlen($prefix));
    41	        $parts    = explode('\\', $relative);
    42	        // Папки в проекте в нижнем регистре: controllers/, config/
    43	        if (isset($parts[0])) {
    44	            $parts[0] = strtolower($parts[0]);
    45	        }
    46	        $file = $baseDir . implode('/', $parts) . '.php';
    47	
    48	        if (file_exists($file)) {
    49	            require_once $file;
    50	        }
    51	    }
    52	});
    53	
    54	// ------------------------------------------------------------------
    55	// Загрузка конфига
    56	// ------------------------------------------------------------------
    57	$config = require __DIR__ . '/config/config.php';
    58	
    59	// ------------------------------------------------------------------
    60	// Заголовки безопасности HTTP
    61	// Устанавливаем ДО любого вывода.
    62	// ------------------------------------------------------------------
    63	
    64	// Запрет встраивания в iframe - защита от Clickjacking
    65	header('X-Frame-Options: DENY');
    66	
    67	// Запрет MIME-сниффинга браузером
    68	header('X-Content-Type-Options: nosniff');
    69	
    70	require_once __DIR__ . '/includes/view.php';
    71	
    72	// Принудительное использование HTTPS на 1 год (только для production)
    73	if (!$config['app']['debug']) {
    74	    header('Strict-Transport-Security: max-age=31536000; includeSubDomains');
    75	}
    76	
    77	// Content Security Policy: разрешаем скрипты/стили только с нашего домена
    78	header("Content-Security-Policy: default-src 'self'; style-src 'self' 'unsafe-inline'; script-src 'self'; img-src 'self' data:;");
    79	
    80	// Запрет отправки Referer на другие домены
    81	header('Referrer-Policy: same-origin');
    82	
    83	// ------------------------------------------------------------------
    84	// Инициализация сессии с безопасными параметрами
    85	// ------------------------------------------------------------------
    86	if (session_status() === PHP_SESSION_NONE) {
    87	    $sess = $config['session'];
    88	
    89	    session_set_cookie_params([
    90	        'lifetime' => $sess['lifetime'],
    91	        'path'     => '/',
    92	        'domain'   => '',
    93	        'secure'   => $sess['secure'],
    94	        'httponly' => $sess['httponly'],
    95	        'samesite' => $sess['samesite'],
    96	    ]);
    97	
    98	    session_start();
    99	
   100	    // Защита от Session Fixation: обновляем ID сессии при каждом запуске
   101	    // (в production делать только при логине, здесь для простоты)
   102	}
   103	
   104	// ------------------------------------------------------------------
   105	// Определение языка интерфейса
   106	// ------------------------------------------------------------------
   107	$allowedLocales   = $config['app']['locales'];
   108	$locale           = $_GET['lang'] ?? $_SESSION['locale'] ?? $config['app']['default_locale'];
   109	
   110	if (!in_array($locale, $allowedLocales, true)) {
   111	    $locale = $config['app']['default_locale'];
   112	}
   113	$_SESSION['locale'] = $locale;
   114	
   115	// ------------------------------------------------------------------
   116	// Маршрутизация
   117	// ------------------------------------------------------------------
   118	
   119	// Получаем чистый URI без query string и нормализуем
   120	$requestUri    = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH) ?: '/';
   121	$requestMethod = strtoupper($_SERVER['REQUEST_METHOD'] ?? 'GET');
   122	
   123	// Сайт в подпапке (например /shop/ в htdocs) - убираем префикс из URI
   124	$scriptDir = str_replace('\\', '/', dirname($_SERVER['SCRIPT_NAME'] ?? ''));
   125	if ($scriptDir !== '/' && $scriptDir !== '' && str_starts_with($requestUri, $scriptDir)) {
   126	    $requestUri = substr($requestUri, strlen($scriptDir)) ?: '/';
   127	}
   128	if ($requestUri === '/index.php') {
   129	    $requestUri = '/';
   130	}
   131	$requestUri = '/' . trim($requestUri, '/');
   132	if ($requestUri !== '/') {
   133	    $requestUri = rtrim($requestUri, '/');
   134	}
   135	
   136	// Разбиваем URI на сегменты: "/product/42" → ['product', '42']
   137	$segments = array_values(array_filter(explode('/', $requestUri)));
   138	
   139	// Первый сегмент - «раздел», второй - параметр
   140	$section = $segments[0] ?? '';
   141	$param   = $segments[1] ?? '';
   142	$subParam = $segments[2] ?? '';
   143	
   144	// ------------------------------------------------------------------
   145	// Таблица маршрутов
   146	// ------------------------------------------------------------------
   147	
   148	try {
   149	
   150	    // ---- GET / (главная страница) --------------------------------
   151	    if ($requestUri === '/' && $requestMethod === 'GET') {
   152	        $controller = new App\Controllers\ProductController();
   153	        $products   = $controller->getProductsByCategory(0, $locale);
   154	        $categories = $controller->getCategories($locale);
   155	        render('home', [
   156	            'products' => $products,
   157	            'categories' => $categories,
   158	            'pageTitle' => 'Quattro - домашняя мебель',
   159	            'pageDescription' => 'Quattro: стильная домашняя мебель для спальни, гостиной, кухни и домашнего офиса с онлайн-конфигуратором.',
   160	            'bodyClass' => 'page-home',
   161	        ]);
   162	        exit;
   163	    }
   164	
   165	    // ---- GET /catalog и GET /catalog/{categoryId} -----------------
   166	    if ($section === 'catalog' && $requestMethod === 'GET') {
   167	        $controller = new App\Controllers\ProductController();
   168	
   169	        if ($param === '') {
   170	            $categories = $controller->getCategories($locale);
   171	            $products   = $controller->getProductsByCategory(0, $locale);
   172	            render('catalog/index', [
   173	                'categories' => $categories,
   174	                'products'   => $products,
   175	                'pageTitle' => 'Каталог',
   176	                'pageDescription' => 'Категории домашней мебели Quattro: спальня, гостиная, кухня и другие комнаты.',
   177	                'bodyClass' => 'page-catalog',
   178	            ]);
   179	            exit;
   180	        }
   181	
   182	        $categoryId = (int)$param;
   183	        $category   = $controller->getCategoryById($categoryId, $locale);
   184	
   185	        if ($category === null) {
   186	            http_response_code(404);
   187	            render('errors/404', ['pageTitle' => '404 - Страница не найдена']);
   188	            exit;
   189	        }
   190	
   191	        $products = $controller->getProductsByCategory($categoryId, $locale);
   192	        render('catalog/category', [
   193	            'category' => $category,
   194	            'products' => $products,
   195	            'pageTitle' => $category['name'] . ' - каталог',
   196	            'pageDescription' => 'Каталог товаров категории ' . $category['name'],
   197	            'bodyClass' => 'page-catalog',
   198	        ]);
   199	        exit;
   200	    }
   201	
   202	    // ---- GET /product/{id|slug} - карточка товара ----------------
   203	    if ($section === 'product' && $requestMethod === 'GET') {
   204	        $controller = new App\Controllers\ProductController();
   205	        $product    = ctype_digit((string)$param)
   206	            ? $controller->getProductById((int)$param, $locale)
   207	            : $controller->getProductBySlug((string)$param, $locale);
   208	
   209	        if ($product === null) {
   210	            http_response_code(404);
   211	            render('errors/404', ['pageTitle' => '404 - Страница не найдена']);
   212	            exit;
   213	        }
   214	
   215	        $description = mb_substr(strip_tags($product['description'] ?? ''), 0, 160);
   216	        $allProducts = $controller->getProductsByCategory(0, $locale);
   217	        render('product/card', [
   218	            'product' => $product,
   219	            'configurableProducts' => filter_configurable_products($allProducts),
   220	            'pageTitle' => $product['name'] . ' - Quattro',
   221	            'pageDescription' => $description,
   222	            'bodyClass' => 'page-product',
   223	            'extraCss' => ['/public/css/configurator.css'],
   224	            'extraJs' => ['/public/js/configurator.js'],
   225	        ]);
   226	        exit;
   227	    }
   228	
   229	    // ---- GET /api/product/{id} - JSON для AJAX -------------------
   230	    if ($section === 'api' && $param === 'product' && $requestMethod === 'GET') {
   231	        $productId  = (int)$subParam;
   232	        $controller = new App\Controllers\ProductController();
   233	        $controller->apiGetProduct($productId);
   234	        exit;
   235	    }
   236	
   237	    // ---- GET /api/csrf-token - выдача CSRF-токена ----------------
   238	    if ($section === 'api' && $param === 'csrf-token' && $requestMethod === 'GET') {
   239	        header('Content-Type: application/json; charset=utf-8');
   240	        $controller = new App\Controllers\OrderController();
   241	        $token      = $controller->generateCsrfToken();
   242	        echo json_encode(['token' => $token], JSON_UNESCAPED_UNICODE);
   243	        exit;
   244	    }
   245	
   246	    // ---- POST /api/order/create - создание заказа ----------------
   247	    if ($section === 'api' && $param === 'order' && $subParam === 'create'
   248	        && $requestMethod === 'POST') {
   249	        $controller = new App\Controllers\OrderController();
   250	        $controller->createOrder();
   251	        exit;
   252	    }
   253	
   254	    // ---- POST /api/lead/create - заявка на КП --------------------
   255	    if ($section === 'api' && $param === 'lead' && $subParam === 'create'
   256	        && $requestMethod === 'POST') {
   257	        $controller = new App\Controllers\LeadController();
   258	        $controller->createLead();
   259	        exit;
   260	    }
   261	
   262	    // ---- GET /order/success/{orderId} - страница «Заказ принят» --
   263	    if ($section === 'order' && $param === 'success' && $requestMethod === 'GET') {
   264	        $orderId    = (int)$subParam;
   265	        $controller = new App\Controllers\OrderController();
   266	        $controller->showSuccess($orderId);
   267	        exit;
   268	    }
   269	
   270	    // ---- GET /privacy - политика конфиденциальности ------------
   271	    if ($section === 'privacy' && $requestMethod === 'GET') {
   272	        render('pages/privacy', [
   273	            'pageTitle' => 'Политика конфиденциальности - Quattro',
   274	            'pageDescription' => 'Как Quattro обрабатывает персональные данные в формах заявки и заказа.',
   275	            'bodyClass' => 'page-privacy',
   276	        ]);
   277	        exit;
   278	    }
   279	
   280	    // ---- GET /cart - корзина ------------------------------------
   281	    if ($section === 'cart' && $requestMethod === 'GET') {
   282	        render('cart', [
   283	            'pageTitle' => 'Корзина - Quattro',
   284	            'pageDescription' => 'Оформление заказа через конфигуратор Quattro.',
   285	            'bodyClass' => 'page-cart',
   286	        ]);
   287	        exit;
   288	    }
   289	
   290	    // ---- 404 - ничего не совпало --------------------------------
   291	    http_response_code(404);
   292	    render('errors/404', ['pageTitle' => '404 - Страница не найдена']);
   293	
   294	} catch (Throwable $e) {
   295	    // Глобальный обработчик непойманных исключений
   296	    // В debug-режиме показываем детали, в production - только общее сообщение
   297	    if ($config['app']['debug']) {
   298	        http_response_code(500);
   299	        echo '<pre style="background:#1a1d27;color:#ef4444;padding:20px;font-family:monospace;">';
   300	        echo '<strong>Ошибка:</strong> ' . htmlspecialchars($e->getMessage()) . "\n\n";
   301	        echo '<strong>Файл:</strong> '   . htmlspecialchars($e->getFile()) . ':' . $e->getLine() . "\n\n";
   302	        echo '<strong>Трассировка:</strong>' . "\n" . htmlspecialchars($e->getTraceAsString());
   303	        echo '</pre>';
   304	    } else {
   305	        // Логируем в файл (не показываем детали пользователю)
   306	        error_log('[' . date('Y-m-d H:i:s') . '] ' . $e->getMessage() . ' in ' . $e->getFile() . ':' . $e->getLine());
   307	        http_response_code(500);
   308	        render('errors/500', ['pageTitle' => '500 - Ошибка сервера']);
   309	    }
   310	}

     1	<?php
     2	/**
     3	 * router.php - точка входа для встроенного сервера PHP (php -S).
     4	 * Эмулирует mod_rewrite: статика отдаётся как есть, остальное → index.php.

     6	declare(strict_types=1);
     7	
     8	$uri = urldecode(parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH) ?? '/');
     9	
    10	if ($uri !== '/' && is_file(__DIR__ . $uri)) {
    11	    return false;
    12	}
    13	
    14	require __DIR__ . '/index.php';

     1	# =============================================================

     5

     9	
    10	# ------------------------------------------------------------------
    11	# ВКЛЮЧАЕМ MOD_REWRITE (перенаправление всех URL на index.php)
    12	# ------------------------------------------------------------------
    13	<IfModule mod_rewrite.c>
    14	    RewriteEngine On
    15	
    16	    # Базовый путь - если сайт в подпапке (например /furniture_platform/),

    30	# ------------------------------------------------------------------

     8	 *  - PDO::ATTR_EMULATE_PREPARES = false - нативные prepared statements

    10	 *  - PDO::ATTR_DEFAULT_FETCH_MODE = FETCH_ASSOC - удобный формат по умолчанию

    13	declare(strict_types=1);
    14	
    15	namespace App\Config;
    16	
    17	use PDO;
    18	use PDOException;
    19	use RuntimeException;

    21	final class Database
    22	{
    23	    private static ?PDO $instance = null;
    24	
    25	    /**
    26	     * Запрещаем прямое создание экземпляров - только через getInstance().
    27	     */

    30	
    31	    /**
    32	     * Возвращает единственный экземпляр PDO-соединения (паттерн Singleton).
    33	     * При первом вызове создаёт соединение, при последующих - возвращает кешированное.
    34	     *
    35	     * @throws RuntimeException если соединение не удалось установить
    36	     */
    37	    public static function getInstance(): PDO
    38	    {
    39	        if (self::$instance === null) {
    40	            $config = require __DIR__ . '/config.php';
    41	            $db     = $config['database'];
    42	
    43	            $dsn = sprintf(
    44	                'mysql:host=%s;port=%d;dbname=%s;charset=utf8mb4',
    45	                $db['host'],
    46	                $db['port'],
    47	                $db['name']
    48	            );
    49

    51	                PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,

    57	                (defined('Pdo\Mysql::ATTR_INIT_COMMAND')
    58	                    ? \Pdo\Mysql::ATTR_INIT_COMMAND
    59	                    : PDO::MYSQL_ATTR_INIT_COMMAND) => "SET NAMES utf8mb4 COLLATE utf8mb4_unicode_ci",

    71	                );
    72	            }
    73	        }

    74	    /**
    75	     * Генерирует криптографически стойкий CSRF-токен и сохраняет его
    76	     * в сессии. Возвращает токен для встраивания в HTML-форму.
    77	     *
    78	     * @return string  32-байтный hex-токен
    79	     */
    80	    public function generateCsrfToken(): string
    81	    {
    82	        // random_bytes() - CSPRNG, безопаснее mt_rand() / uniqid()
    83	        $token = bin2hex(random_bytes(32));
    84	        $_SESSION['csrf_token'] = $token;
    85	        return $token;
    86	    }

    88	    /**
    89	     * Проверяет CSRF-токен из запроса против значения в сессии.
    90	     * Использует hash_equals() для защиты от timing-атак.
    91	     *
    92	     * @param string $tokenFromRequest  Токен, переданный клиентом
    93	     * @throws RuntimeException         Если токен невалиден или отсутствует
    94	     */
    95	    private function validateCsrfToken(string $tokenFromRequest): void
    96	    {
    97	        $sessionToken = $_SESSION['csrf_token'] ?? '';
    98	
    99	        if (
   100	            empty($sessionToken) ||
   101	            !hash_equals($sessionToken, $tokenFromRequest)
   102	        ) {
   103	            throw new RuntimeException('Недействительный CSRF-токен.', 403);
   104	        }
   105	
   106	        // Токен одноразовый: инвалидируем после успешной проверки,
   107	        // чтобы повторная отправка той же формы не прошла проверку.
   108	        unset($_SESSION['csrf_token']);
   109	    }

   115	    /**
   116	     * POST /api/order/create
   117	     *
   118	     * Принимает тело запроса в формате JSON:
   119	     * {
   120	     *   "csrf_token": "abc123...",
   121	     *   "items": [
   122	     *     {
   123	     *       "product_id": 1,
   124	     *       "quantity": 1,
   125	     *       "selected_options": [3, 7, 9]   // массив ID выбранных опций
   126	     *     }
   127	     *   ],
   128	     *   "delivery": {
   129	     *     "name":    "Иван Иванов",
   130	     *     "phone":   "+77001234567",
   131	     *     "city":    "Алматы",
   132	     *     "address": "ул. Абая, 10, кв. 5",
   133	     *     "comment": ""
   134	     *   }
   135	     * }
   136	     *
   137	     * Возвращает JSON:
   138	     *   { "success": true, "order_id": 42, "total_price": 985000 }
   139	     *   или
   140	     *   { "success": false, "error": "Текст ошибки" }
   141	     */
   142	    public function createOrder(): void
   143	    {
   144	        header('Content-Type: application/json; charset=utf-8');
   145	
   146	        // Разрешаем только POST-запросы
   147	        if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
   148	            http_response_code(405);
   149	            echo json_encode(['success' => false, 'error' => 'Метод не разрешён.'],
   150	                             JSON_UNESCAPED_UNICODE);
   151	            return;
   152	        }
   153	
   154	        try {
   155	            // --- 1. Чтение и первичный разбор JSON-тела запроса ---
   156	            $rawBody = file_get_contents('php://input');
   157	            $data    = json_decode($rawBody, true);
   158	
   159	            if (json_last_error() !== JSON_ERROR_NONE || !is_array($data)) {
   160	                throw new InvalidArgumentException('Некорректный формат JSON в теле запроса.');
   161	            }
   162	
   163	            // --- 2. Проверка CSRF-токена ---
   164	            $csrfToken = $this->sanitizeString($data['csrf_token'] ?? '');
   165	            $this->validateCsrfToken($csrfToken);
   166	
   167	            // --- 3. Валидация и санитизация входных данных ---
   168	            $items    = $this->validateItems($data['items'] ?? []);
   169	            $delivery = $this->validateDelivery($data['delivery'] ?? []);
   170	
   171	            // --- 4. Расчёт стоимости и запись в БД ---
   172	            $orderId = $this->persistOrder($items, $delivery);
   173	
   174	            http_response_code(201);
   175	            echo json_encode([
   176	                'success'  => true,
   177	                'order_id' => $orderId,
   178	                'message'  => 'Ваш заказ успешно принят. Менеджер свяжется с вами в течение 30 минут.',
   179	            ], JSON_UNESCAPED_UNICODE);
   180	
   181	        } catch (InvalidArgumentException $e) {
   182	            http_response_code(422);
   183	            echo json_encode(['success' => false, 'error' => $e->getMessage()],
   184	                             JSON_UNESCAPED_UNICODE);
   185	
   186	        } catch (RuntimeException $e) {
   187	            http_response_code($e->getCode() ?: 500);
   188	            echo json_encode(['success' => false, 'error' => $e->getMessage()],
   189	                             JSON_UNESCAPED_UNICODE);
   190	
   191	        } catch (\Throwable $e) {
   192	            // Непредвиденная ошибка: логируем, но не раскрываем детали клиенту
   193	            error_log('[OrderController] Unexpected error: ' . $e->getMessage());
   194	            http_response_code(500);
   195	            echo json_encode(['success' => false, 'error' => 'Внутренняя ошибка сервера.'],
   196	                             JSON_UNESCAPED_UNICODE);
   197	        }
   198	    }

   230	    /**
   231	     * Валидирует массив позиций заказа.
   232	     * Каждая позиция: product_id (int > 0), quantity (int 1..MAX_QTY),
   233	     *                 selected_options (array of int)
   234	     *
   235	     * @param mixed $rawItems
   236	     * @return array  Валидированный массив позиций
   237	     * @throws InvalidArgumentException
   238	     */
   239	    private function validateItems(mixed $rawItems): array
   240	    {
   241	        if (!is_array($rawItems) || empty($rawItems)) {
   242	            throw new InvalidArgumentException('Корзина не может быть пустой.');
   243	        }
   244	
   245	        if (count($rawItems) > self::MAX_ITEMS) {
   246	            throw new InvalidArgumentException(
   247	                sprintf('В заказе не может быть более %d позиций.', self::MAX_ITEMS)
   248	            );
   249	        }
   250	
   251	        $validated = [];
   252	
   253	        foreach ($rawItems as $index => $item) {
   254	            if (!is_array($item)) {
   255	                throw new InvalidArgumentException(
   256	                    "Позиция #{$index}: некорректный формат данных."
   257	                );
   258	            }
   259	
   260	            // product_id - целое положительное число
   261	            $productId = filter_var($item['product_id'] ?? 0,
   262	                                    FILTER_VALIDATE_INT,
   263	                                    ['options' => ['min_range' => 1]]);
   264	            if ($productId === false) {
   265	                throw new InvalidArgumentException(
   266	                    "Позиция #{$index}: некорректный идентификатор товара."
   267	                );
   268	            }
   269	
   270	            // quantity - целое от 1 до MAX_QTY
   271	            $quantity = filter_var($item['quantity'] ?? 1,
   272	                                   FILTER_VALIDATE_INT,
   273	                                   ['options' => ['min_range' => 1, 'max_range' => self::MAX_QTY]]);
   274	            if ($quantity === false) {
   275	                throw new InvalidArgumentException(
   276	                    "Позиция #{$index}: некорректное количество."
   277	                );
   278	            }
   279	
   280	            // selected_options - массив целых чисел (может быть пустым)
   281	            $rawOptions = $item['selected_options'] ?? [];
   282	            if (!is_array($rawOptions)) {
   283	                throw new InvalidArgumentException(
   284	                    "Позиция #{$index}: опции должны быть переданы массивом."
   285	                );
   286	            }
   287	
   288	            $optionIds = [];
   289	            foreach ($rawOptions as $optId) {
   290	                $validated_opt = filter_var($optId,
   291	                                            FILTER_VALIDATE_INT,
   292	                                            ['options' => ['min_range' => 1]]);
   293	                if ($validated_opt === false) {
   294	                    throw new InvalidArgumentException(
   295	                        "Позиция #{$index}: некорректный ID опции."
   296	                    );
   297	                }
   298	                $optionIds[] = (int)$validated_opt;
   299	            }
   300	
   301	            $validated[] = [
   302	                'product_id'       => (int)$productId,
   303	                'quantity'         => (int)$quantity,
   304	                'selected_options' => $optionIds,
   305	            ];
   306	        }
   307	
   308	        return $validated;
   309	    }

   311	    /**
   312	     * Валидирует и санитизирует данные доставки.
   313	     *
   314	     * @param mixed $rawDelivery
   315	     * @return array
   316	     * @throws InvalidArgumentException
   317	     */
   318	    private function validateDelivery(mixed $rawDelivery): array
   319	    {
   320	        if (!is_array($rawDelivery)) {
   321	            throw new InvalidArgumentException('Отсутствуют данные доставки.');
   322	        }
   323	
   324	        $name    = $this->sanitizeString($rawDelivery['name']    ?? '');
   325	        $phone   = $this->sanitizeString($rawDelivery['phone']   ?? '');
   326	        $city    = $this->sanitizeString($rawDelivery['city']    ?? '');
   327	        $address = $this->sanitizeString($rawDelivery['address'] ?? '');
   328	        $comment = $this->sanitizeString($rawDelivery['comment'] ?? '');
   329	
   330	        // Обязательные поля
   331	        if (mb_strlen($name) < 2 || mb_strlen($name) > 150) {
   332	            throw new InvalidArgumentException('Укажите корректное имя (2-150 символов).');
   333	        }
   334	
   335	        // Валидация телефона: допускаем форматы +7XXXXXXXXXX или 8XXXXXXXXXX
   336	        if (!preg_match('/^(\+7|8)\d{10}$/', preg_replace('/[\s\-()]/', '', $phone))) {
   337	            throw new InvalidArgumentException('Укажите корректный номер телефона.');
   338	        }
   339	
   340	        if (mb_strlen($city) < 2 || mb_strlen($city) > 100) {
   341	            throw new InvalidArgumentException('Укажите корректное название города.');
   342	        }
   343	
   344	        if (mb_strlen($address) < 5 || mb_strlen($address) > 500) {
   345	            throw new InvalidArgumentException('Укажите корректный адрес доставки.');
   346	        }
   347	
   348	        return [
   349	            'name'    => $name,
   350	            'phone'   => $phone,
   351	            'city'    => $city,
   352	            'address' => $address,
   353	            'comment' => mb_substr($comment, 0, 1000), // ограничиваем длину комментария
   354	        ];
   355	    }

   361	    /**
   362	     * Рассчитывает стоимость позиций заказа и записывает заказ в БД
   363	     * в рамках одной транзакции InnoDB (гарантия ACID).
   364	     *
   365	     * Алгоритм расчёта стоимости:
   366	     *   item_price = product.base_price + SUM(selected option.price_modifier)
   367	     *   total      = SUM(item_price * quantity) по всем позициям
   368	     *
   369	     * @param array $items     Валидированные позиции заказа
   370	     * @param array $delivery  Валидированные данные доставки
   371	     * @return int             ID созданного заказа
   372	     * @throws RuntimeException|InvalidArgumentException
   373	     */
   374	    private function persistOrder(array $items, array $delivery): int
   375	    {
   376	        $this->db->beginTransaction();
   377	
   378	        try {
   379	            $totalOrderPrice = 0.0;
   380	            $processedItems  = [];
   381	
   382	            foreach ($items as $item) {
   383	                $productId = $item['product_id'];
   384	                $quantity  = $item['quantity'];
   385	                $optionIds = $item['selected_options'];
   386	
   387	                // --- Получаем базовую цену товара из БД ---
   388	                // Это критически важно: нельзя доверять цене, переданной клиентом -
   389	                // клиентская часть хранит отображаемые цены, реальные берём из БД.
   390	                $productStmt = $this->db->prepare(
   391	                    'SELECT id, base_price FROM products WHERE id = :id AND active = 1 LIMIT 1'
   392	                );
   393	                $productStmt->execute([':id' => $productId]);
   394	                $product = $productStmt->fetch();
   395	
   396	                if ($product === false) {
   397	                    throw new InvalidArgumentException(
   398	                        "Товар с ID {$productId} не найден или недоступен."
   399	                    );
   400	                }
   401	
   402	                $itemPrice      = (float)$product['base_price'];
   403	                $selectedOptions = [];
   404	
   405	                // --- Получаем и суммируем price_modifier выбранных опций ---
   406	                if (!empty($optionIds)) {
   407	                    $placeholders = implode(',', array_fill(0, count($optionIds), '?'));
   408	
   409	                    $optStmt = $this->db->prepare("
   410	                        SELECT id, option_name_ru, price_modifier
   411	                        FROM product_options
   412	                        WHERE id IN ({$placeholders})
   413	                          AND product_id = ?
   414	                    ");
   415	
   416	                    // Передаём: сначала ID опций, затем ID продукта
   417	                    $optStmt->execute([...$optionIds, $productId]);
   418	                    $fetchedOptions = $optStmt->fetchAll();
   419	
   420	                    // Защита: количество найденных опций должно совпадать с запрошенными
   421	                    if (count($fetchedOptions) !== count($optionIds)) {
   422	                        throw new InvalidArgumentException(
   423	                            "Одна или несколько выбранных опций для товара ID {$productId} не существуют."
   424	                        );
   425	                    }
   426	
   427	                    foreach ($fetchedOptions as $opt) {
   428	                        $itemPrice += (float)$opt['price_modifier'];
   429	                        $selectedOptions[] = [
   430	                            'option_id'      => (int)$opt['id'],
   431	                            'option_name_ru' => $opt['option_name_ru'],
   432	                            'price_modifier' => (float)$opt['price_modifier'],
   433	                        ];
   434	                    }
   435	                }
   436	
   437	                $lineTotal        = $itemPrice * $quantity;
   438	                $totalOrderPrice += $lineTotal;
   439	
   440	                $processedItems[] = [
   441	                    'product_id'           => $productId,
   442	                    'quantity'             => $quantity,
   443	                    'price'                => $itemPrice,
   444	                    'selected_options_json' => json_encode(
   445	                        $selectedOptions,
   446	                        JSON_UNESCAPED_UNICODE
   447	                    ),
   448	                ];
   449	            }
   450	
   451	            // --- Записываем заголовок заказа в таблицу orders ---
   452	            $deliveryJson = json_encode($delivery, JSON_UNESCAPED_UNICODE);
   453	
   454	            $userId = $_SESSION['user_id'] ?? null; // null если гостевой заказ
   455	
   456	            $orderStmt = $this->db->prepare("
   457	                INSERT INTO orders (user_id, total_price, status, delivery_info, created_at)
   458	                VALUES (:user_id, :total_price, 'new', :delivery_info, NOW())
   459	            ");
   460	            $orderStmt->execute([
   461	                ':user_id'       => $userId,
   462	                ':total_price'   => round($totalOrderPrice, 2),
   463	                ':delivery_info' => $deliveryJson,
   464	            ]);
   465	
   466	            $orderId = (int)$this->db->lastInsertId();
   467	
   468	            // --- Записываем позиции заказа в таблицу order_items ---
   469	            $itemStmt = $this->db->prepare("
   470	                INSERT INTO order_items
   471	                    (order_id, product_id, selected_options_json, quantity, price)
   472	                VALUES
   473	                    (:order_id, :product_id, :options_json, :quantity, :price)
   474	            ");
   475	
   476	            foreach ($processedItems as $processed) {
   477	                $itemStmt->execute([
   478	                    ':order_id'    => $orderId,
   479	                    ':product_id'  => $processed['product_id'],
   480	                    ':options_json' => $processed['selected_options_json'],
   481	                    ':quantity'    => $processed['quantity'],
   482	                    ':price'       => round($processed['price'], 2),
   483	                ]);
   484	            }
   485	
   486	            // Всё прошло успешно - фиксируем транзакцию
   487	            $this->db->commit();
   488	
   489	            return $orderId;
   490	
   491	        } catch (\Throwable $e) {
   492	            // При любой ошибке откатываем транзакцию - частичных заказов не будет
   493	            $this->db->rollBack();
   494	            throw $e;
   495	        }
   496	    }

   498	    /**
   499	     * Возвращает данные заказа по его ID (для страницы подтверждения).
   500	     */
   501	    private function getOrderById(int $orderId): ?array
   502	    {
   503	        $stmt = $this->db->prepare(
   504	            'SELECT id, total_price, status, delivery_info, created_at
   505	             FROM orders WHERE id = :id LIMIT 1'
   506	        );
   507	        $stmt->execute([':id' => $orderId]);
   508	        $order = $stmt->fetch();
   509	
   510	        return $order !== false ? $order : null;
   511	    }

   517	    /**
   518	     * Очищает строку от пробелов и экранирует HTML-спецсимволы.
   519	     * htmlspecialchars с ENT_QUOTES экранирует как " так и ' -
   520	     * защита от XSS при последующем выводе в HTML-контексте.
   521	     *
   522	     * @param mixed  $value   Входное значение
   523	     * @param int    $maxLen  Максимальная длина после обрезки
   524	     * @return string
   525	     */
   526	    private function sanitizeString(mixed $value, int $maxLen = 500): string
   527	    {
   528	        if (!is_string($value) && !is_numeric($value)) {
   529	            return '';
   530	        }
   531	
   532	        $clean = trim((string)$value);
   533	        $clean = mb_substr($clean, 0, $maxLen);
   534	
   535	        // ENT_QUOTES: экранируем и двойные, и одинарные кавычки
   536	        return htmlspecialchars($clean, ENT_QUOTES | ENT_HTML5, 'UTF-8');
   537	    }

   123	    /**
   124	     * @param array<string, mixed> $data
   125	     * @return array<string, string|null>
   126	     */
   127	    private function validateLead(array $data): array
   128	    {
   129	        $name = $this->sanitizeString($data['name'] ?? '', 150);
   130	        if (mb_strlen($name) < 2 || mb_strlen($name) > 150) {
   131	            throw new InvalidArgumentException('Укажите корректное имя (2-150 символов).');
   132	        }
   133	
   134	        $email = trim((string)($data['email'] ?? ''));
   135	        if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
   136	            throw new InvalidArgumentException('Укажите корректный email.');
   137	        }
   138	        $email = $this->sanitizeString($email, 255);
   139	
   140	        $phoneRaw = trim((string)($data['phone'] ?? ''));
   141	        $phoneDigits = preg_replace('/\D/', '', $phoneRaw) ?? '';
   142	        if (strlen($phoneDigits) < 10) {
   143	            throw new InvalidArgumentException('Укажите корректный номер телефона (минимум 10 цифр).');
   144	        }
   145	        $phone = $this->sanitizeString($phoneRaw, 50);
   146	
   147	        $organization = $this->sanitizeString($data['organization'] ?? '', 200);
   148	        $organization = $organization !== '' ? $organization : null;
   149	
   150	        $comment = $this->sanitizeString($data['comment'] ?? '', 2000);
   151	        $comment = $comment !== '' ? $comment : null;
   152	
   153	        $source = $this->sanitizeString($data['source'] ?? 'home', 20);
   154	        if (!in_array($source, self::ALLOWED_SOURCES, true)) {
   155	            $source = 'home';
   156	        }
   157	
   158	        return [
   159	            'email'        => $email,
   160	            'name'         => $name,
   161	            'phone'        => $phone,
   162	            'organization' => $organization,
   163	            'comment'      => $comment,
   164	            'source'       => $source,
   165	        ];
   166	    }

   175	            VALUES (:email, :name, :phone, :organization, :comment, :source, 'new', NOW())

   190	    private function sanitizeString(mixed $value, int $maxLen = 500): string
   191	    {
   192	        if (!is_string($value) && !is_numeric($value)) {
   193	            return '';
   194	        }
   195	
   196	        $clean = trim((string)$value);
   197	        $clean = mb_substr($clean, 0, $maxLen);
   198	
   199	        return htmlspecialchars($clean, ENT_QUOTES | ENT_HTML5, 'UTF-8');
   200	    }
